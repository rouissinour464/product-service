package com.store.productservice.service;
import com.store.productservice.dto.ProductRequest;import com.store.productservice.dto.ProductResponse;import com.store.productservice.entity.Product;import com.store.productservice.exception.ResourceNotFoundException;import com.store.productservice.repository.ProductRepository;import org.springframework.stereotype.Service;import java.time.LocalDateTime;import java.util.List;import java.util.stream.Collectors;
@Service
public class ProductService { private final ProductRepository repository; public ProductService(ProductRepository repository){this.repository=repository;} public List<ProductResponse> getAll(){return repository.findAll().stream().map(this::toResponse).collect(Collectors.toList());} public ProductResponse getById(Long id){Product p=repository.findById(id).orElseThrow(() -> new ResourceNotFoundException("Produit introuvable: id="+id)); return toResponse(p);} public ProductResponse create(ProductRequest req){ LocalDateTime now = LocalDateTime.now(); LocalDateTime dc = req.getDateCreation(); if (dc == null || dc.isAfter(now)) { dc = now; } Product p=new Product(req.getNom(),req.getPrix(),req.getQuantite(),dc); return toResponse(repository.save(p)); } public ProductResponse update(Long id, ProductRequest req){ Product p=repository.findById(id).orElseThrow(() -> new ResourceNotFoundException("Produit introuvable: id="+id)); p.setNom(req.getNom()); p.setPrix(req.getPrix()); p.setQuantite(req.getQuantite()); LocalDateTime now = LocalDateTime.now(); LocalDateTime dc = req.getDateCreation(); p.setDateCreation((dc == null || dc.isAfter(now)) ? now : dc); return toResponse(repository.save(p)); } public void delete(Long id){ if(!repository.existsById(id)) throw new ResourceNotFoundException("Produit introuvable: id="+id); repository.deleteById(id);} public List<ProductResponse> searchByName(String name){return repository.findByNomContainingIgnoreCase(name).stream().map(this::toResponse).collect(Collectors.toList());} private ProductResponse toResponse(Product p){return new ProductResponse(p.getId(),p.getNom(),p.getPrix(),p.getQuantite(),p.getDateCreation());} }
